# 1. 费马小定理+快速幂求乘法逆元

```cpp
#include <bits/stdc++.h>  
using namespace std;  
typedef long long ll;  
// 快速幂  
ll qpow(ll a, ll b, ll p) {  
    ll res = 1;  
    while (b) {  
        if (b & 1) res = res * a % p;  
        a = a * a % p;  
        b >>= 1;  
    }  
    return res;  
}  
// 费马小定理求逆元  
ll inv(ll a, ll p) {  
    return qpow(a, p - 2, p); // p 必须为质数  
}  
int main() {  
    ll a = 3, p = 7;  
    cout << inv(a, p) << endl; // 输出 5，因为 3*5 ≡ 1 (mod 7)}
```
